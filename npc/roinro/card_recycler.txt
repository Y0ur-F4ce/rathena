m_office,60,34,6	script	Card Recycler	850,{
	mes "[Card Recycler]";
	mes "Hi, what can I do for you?";
	next;
	switch(select("What are you doing?:Trade in cards:Point exchange (^0055FF"+getd(.Points$)+"^000000):Leave")) {
	case 1:
		mes "[Card Recycler]";
		mes "Do you find that you've got";
		mes "useless cards lying around?";
		mes "I'll be glad to take them off";
		mes "your hands!";
		next;
		mes "[Card Recycler]";
		mes "For each card you give me, I'll check the corresponding monster's level.";
		mes "For each level that monster is, you'll get 1 point.";
		mes "You can trade in 10 points for a Roin Coin!";
		if (.Level) {
			next;
			mes "[Card Recycler]";
			mes "However, the cards must be dropped by monsters whose level";
			mes "is at LEAST " + .Level+".";
		}
		mes "How does that sound?";
		emotion e_cash;
		close;
	case 2:
		mes "[Card Recycler]";
		mes "Select the cards you";
		mes "want to trade in.";
		if (.Level) {
			mes "They must be dropped";
			mes "by monsters of level";
			mes .Level+" and above.";
		}
		deletearray @sold_nameid[0],getarraysize(@sold_nameid);
		close2;
		callshop "card_shop",2;
		npcshopattach "card_shop";
		end;
	case 3:
		if (#Card_Points < 10) {
			mes "[Card Recycler]";
			mes "You don't have enough points to do anything!";
			close;
		}
		if (!checkweight(1231,2)) {
			mes "Sorry, but you're carrying too many items. Empty some of your inventory and then come back.";
			close3;
			end;
		}
		mes "[Card Recycler]";
		mes "You have ^0055FF"+getd(.Points$)+"^000000 Point"+((getd(.Points$) == 1)?".":"s.");
		mes "You can get 1 Roin Coin per 10 Points! How many Roin Coins would you like to withdraw?";
		switch(select("As many as I can!","A specific amount","Cancel")) {
			case 1: // As many as possible
				.@roin_coins = floor(#Card_Points/10,1);
				.@remainder = #Card_Points%10;
				mes "Withdraw ^0055FF" + .@roin_coins + "^000000 Roin Coins? You will have ^0055FF" + .@remainder + "^000000 Points remaining.";
				if (select("GIVE ME ROIN COINS","Cancel") == 2) {
					close2;
					end;
				}
				getitem 250000, .@roin_coins;
				#Card_Points = .@remainder;
				soundeffect "se_cash_provider.wav",0; // Reusing the sound effect from the Enigmatic Slot Machine
				close;
				break;
			case 2:
				clear;
				mes "[Card Recycler]";
				mes "How many would you like to receive?";
				mes "(Maximum: ^0055FF" + floor(#Card_Points/10,1) + "^000000 Roin Coins)";
				input .@roin_coins_amt;
				if (.@roin_coins_amt <= 0) {
					close;
				}
				if (.@roin_coins_amt > floor(#Card_Points/10,1)) {
					clear;
					mes "[Card Recycler]";
					mes "You don't enough points to withdraw that many Roin Coins.";
					close;
				}
				.@leftover = #Card_Points - (10*.@roin_coins_amt);
				mes "Withdraw ^0055FF" + .@roin_coins_amt + "^000000 Roin Coins? You will have ^0055FF" + .@leftover + "^000000 Points remaining.";
				if (select("GIVE ME ROIN COINS","Cancel") == 2) {
					close2;
					end;
				}
				getitem 250000, .@roin_coins_amt;
				#Card_Points = .@leftover;
				soundeffect "se_cash_provider.wav",0; // Reusing the sound effect from the Enigmatic Slot Machine
				close;
				break;
			case 3:
				close;
				break;
		}
		end;
	case 4:
		mes "[Card Recycler]";
		mes "*yawn*";
		mes "See you later!";
		emotion e_yawn;
		close;		
	}

OnSellItem:
	mes "Cards to sell:";
	mes "-----------------------------------";
	for(set .@i,0; .@i<getarraysize(@sold_nameid); set .@i,.@i+1)
		if ((@sold_nameid[.@i] > 4000 && @sold_nameid[.@i] < 4700) || 
		(@sold_nameid[.@i] > 27011 && @sold_nameid[.@i] < 27363) || 
		(@sold_nameid[.@i] > 31016 && @sold_nameid[.@i] < 31027) ||
		(@sold_nameid[.@i] > 300000 && @sold_nameid[.@i] < 300282) ||
		(@sold_nameid[.@i] > 300358 && @sold_nameid[.@i] < 300383)
		) {
			if (.Level) {
				//query_sql("SELECT `level` FROM `mob_db_re` WHERE `drop8_item` = '"+getiteminfo(@sold_nameid[.@i], ITEMINFO_AEGISNAME)+"'",.@lv);
				query_sql(
					"SELECT `level` FROM `mob_db_re` "
					+ "WHERE `drop1_item`='" + getiteminfo(@sold_nameid[.@i], ITEMINFO_AEGISNAME) + "'"
					+ " OR `drop2_item`='" + getiteminfo(@sold_nameid[.@i], ITEMINFO_AEGISNAME) + "'"
					+ " OR `drop3_item`='" + getiteminfo(@sold_nameid[.@i], ITEMINFO_AEGISNAME) + "'"
					+ " OR `drop4_item`='" + getiteminfo(@sold_nameid[.@i], ITEMINFO_AEGISNAME) + "'"
					+ " OR `drop5_item`='" + getiteminfo(@sold_nameid[.@i], ITEMINFO_AEGISNAME) + "'"
					+ " OR `drop6_item`='" + getiteminfo(@sold_nameid[.@i], ITEMINFO_AEGISNAME) + "'"
					+ " OR `drop7_item`='" + getiteminfo(@sold_nameid[.@i], ITEMINFO_AEGISNAME) + "'"
					+ " OR `drop8_item`='" + getiteminfo(@sold_nameid[.@i], ITEMINFO_AEGISNAME) + "'"
					+ " ORDER BY `level` ASC LIMIT 1",
					.@lv
				);
				if (.@lv < .Level) {
					dispbottom getitemname(@sold_nameid[.@i])+" is under the minimum level.";
					continue;
				}
			}
			set .@card_id[getarraysize(.@card_id)], @sold_nameid[.@i];
			set .@card_amt[getarraysize(.@card_amt)], @sold_quantity[.@i];
			set .@mvp, compare(.MVP$,""+@sold_nameid[.@i]);
			mes ((.@mvp)?"  ^FF0000":"  ^777777")+@sold_quantity[.@i]+"x "+getitemname(@sold_nameid[.@i])+"^000000";
			if (@sold_nameid[.@i] == 27030) { //Check if the card is Fire Bug card (very low HP to level ratio)
				set .@card_total, .@card_total+(@sold_quantity[.@i]*33);
			}
			else {
				set .@card_total, .@card_total+(@sold_quantity[.@i]*.@lv);
			}
		}
	deletearray @sold_nameid[0], getarraysize(@sold_nameid);
	deletearray @sold_quantity[0], getarraysize(@sold_quantity);
	if (!.@card_id) {
		mes "  ^777777(none)^000000";
		emotion e_swt;
		close;
	}
	mes " ";
	mes "---------- Total: ^0055FF"+.@card_total+" pt.^000000 -------";
	next;
	if(select(" > ^0055FFComplete trade...^000000: > ^777777Cancel^000000") == 2) {
		mes "[Card Recycler]";
		mes "Oh, okay...";
		emotion e_hmm;
		close;
	}
	for(set .@i,0; .@i<getarraysize(.@card_id); set .@i,.@i+1)
		delitem .@card_id[.@i],.@card_amt[.@i];
	setd .Points$, getd(.Points$)+.@card_total;
	mes "[Card Recycler]";
	mes "All done!";
	emotion e_ho;
	close;

OnBuyItem:
	for(set .@i,0; .@i<getarraysize(@bought_nameid); set .@i,.@i+1)
		for(set .@j,0; .@j<getarraysize(.Shop); set .@j,.@j+2)
			if (@bought_nameid[.@i] == .Shop[.@j]) {
				set .@cost, .@cost+(.Shop[.@j+1]*@bought_quantity[.@i]);
				break;
			}
	if (.@cost > getd(.Points$)) {
		mes "[Card Recycler]";
		mes "You don't have enough Points.";
		emotion e_omg;
	}
	else {
		mes "Items purchased:";
		mes "-----------------------------------";
		for(set .@i,0; .@i<getarraysize(@bought_nameid); set .@i,.@i+1) {
			getitem @bought_nameid[.@i], @bought_quantity[.@i];
			mes "  ^777777"+@bought_quantity[.@i]+"x "+getitemname(@bought_nameid[.@i])+"^000000";
		}
		mes " ";
		mes "---------- Total: ^0055FF"+.@cost+" pt.^000000 -------";
		setd .Points$, getd(.Points$)-.@cost;
		emotion e_cash;
	}
	deletearray @bought_nameid[0], getarraysize(@bought_nameid);
	deletearray @bought_quantity[0], getarraysize(@bought_quantity);
	close;

OnInit:
	set .Level,30;   		// Minimum monster level to trade corresponding cards.
	set .Points$,"#Card_Points";	// Variable to store points.
	setarray .Shop[0],		// Card Shop items: <ID>,<point cost>
	  4001,1,4004,1,4033,1,4196,1,4197,25,4054,50,4174,75,4047,100;
	setarray .Points[0],1,5;	// Points per <normal card>,<MVP card>
	set .MVP$,			// List of MVP cards.
	  "4121,4123,4128,4131,4132,4134,4135,4137,4142,4143,4144,4145,4146,4147,4148,4168,4236,"+
	  "4241,4263,4276,4302,4305,4318,4324,4330,4342,4357,4359,4361,4363,4365,4399,4403,4407";
	
	npcshopdelitem "card_shop",909;
	for(set .@i,0; .@i<getarraysize(.Shop); set .@i,.@i+2)
		npcshopadditem "card_shop",.Shop[.@i],.Shop[.@i+1];
	end;
}
