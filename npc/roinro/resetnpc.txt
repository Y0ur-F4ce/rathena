//===== rAthena Script =======================================
//= Reset NPC
//===== Description: =========================================
//= Resets skills, stats, or both.
//===== Additional Comments: =================================
//= 1.0 First Version
//= 1.1 Optimized for the greater good. [Kisuka]
//= 1.2 Cleaning [Euphy]
//= 1.3 All statuses removed upon skill reset. [Euphy]
//= 1.4 Compressed Script, Added limit use option [Stolao]
//=	Changed set -> setarray, Improved text with F_InsertPlural
//= 1.5 Added sc_end_class to reset related status changes [sader1992]
//============================================================
m_office,44,29,4	script	Reset Man	483,{
	cutin "reset_man1",2;
	//	 		Skills,	Stats,	Both,	Limit
	setarray .@Reset,	5000,	5000,	9000,	0;
	mes "[Reset Man]";
	if(.@Reset[3] && reset_limit > .@Reset[3]) {
		mes "Sorry you can only reset "+callfunc("F_InsertPlural",.@Reset[3],"time")+" in your life.";
		close;
	}
	//mes "I am the Reset Man.";
	mes "*It looks like he wants to punch you*";
	//mes "Reset Stats: "+ callfunc("F_InsertComma",.@Reset[1]) +"z";
	//mes "Reset Skills: "+ callfunc("F_InsertComma",.@Reset[0]) +"z";
	//mes "Reset Both: "+ callfunc("F_InsertComma",.@Reset[2]) +"z";
	if(.@Reset[3]) mes "You may only reset "+callfunc("F_InsertPlural",.@Reset[3],"time")+", so use "+((.@Reset[3]>1)?"them":"it")+" wisely.";
	mes "";
	mes "(Please select the service you want)";
	next;
	set .@i,(select("^FF3355Reset Skills - 5,000 zeny:Reset Stats - 5,000 zeny:Reset Both - 9,000 zeny^000000:Cancel"));
	if(.@i > 3) {close2; cutin "",255; end;};
	mes "[Reset Man]";
	if (Zeny < .@Reset[.@i-1]) {
		mes "You're so poor, you're not even worth punching.";
		//mes "Sorry, you don't have enough Zeny.";
		close3;
	}
	if(.@Reset[3]){
		mes "You can only reset "+callfunc("F_InsertPlural",.@Reset[3],"time")+" in your life, are you sure?";
		if(select("Let me think:That's fine") == 1) close;
	}
	//Double checking with user
	if(.@i == 1) {
		mes "(Reset your skills?)";
		set .@yesno,(select("^FF3355Yes:^000000Cancel"));
		if(.@yesno&1){
			set Zeny, Zeny-.@Reset[.@i-1];
			sc_end_class;
			ResetSkill;
			specialeffect2 EF_HIT2;
			pushpc DIR_SOUTH, 1;
			mes "*He punches you so hard that";
			mes " you forgot your skills*";
			if(.@Reset[3]) set reset_limit,reset_limit + 1;
			cutin "",255;
			close;
		}
		if(.@yesno&2) {close2; cutin "",255; end;}; 
	}
	if(.@i == 2) {
		mes "(Reset your stats?)";
		set .@yesno,(select("^FF3355Yes:^000000Cancel"));
		if(.@yesno&1){
			set Zeny, Zeny-.@Reset[.@i-1];
			ResetStatus;
			specialeffect2 EF_HIT2;
			pushpc DIR_SOUTH, 1;
			mes "*He punches you so hard that";
			mes " you forgot your stats*";
			if(.@Reset[3]) set reset_limit,reset_limit + 1;
			cutin "",255;
			close;
		}
		if(.@yesno&2) {close2; cutin "",255; end;}; 
	}
	if(.@i == 3) {
		mes "(Reset both your stats and skills?)";
		set .@yesno,(select("^FF3355Yes:^000000Cancel"));
		if(.@yesno&1){
			set Zeny, Zeny-.@Reset[.@i-1];
			sc_end_class;
			ResetSkill;
			ResetStatus;
			specialeffect2 EF_HIT2;
			pushpc DIR_SOUTH, 1;
			mes "*He punches you so hard that";
			mes " you forgot your stats and skills*";
			if(.@Reset[3]) set reset_limit,reset_limit + 1;
			cutin "",255;
			close;
		}
		if(.@yesno&2) {close2; cutin "",255; end;}; 
	}
	/*
	if(.@i&1){
		sc_end_class;
		ResetSkill;
	}
	if(.@i&2) ResetStatus;
	mes "*He punches you so hard,";
	mes " you forgot your stats/skills*";
	if(.@Reset[3]) set reset_limit,reset_limit + 1;
	cutin "",255;
	close;
*/
}
