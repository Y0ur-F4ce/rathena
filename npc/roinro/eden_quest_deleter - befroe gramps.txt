	function F_confirm {
		clear;
		mes "[Quest Deletion]";
		mes "Delete " + getarg(0,"this quest") + "? This CANNOT be undone.";
		if(select("Yes:Nevermind") == 2) {
			clear;
			mes "[Quest Deletion]";
			mes "Very well, have a good day.";
			close;
		}
		if(isbegin_quest(getarg(1))) {
			erasequest getarg(1);
			clear;
			mes "[Quest Deletion]";
			mes "It's done. Have a good day.";
			close;
		}
		clear;
		mes "Odd, it doesn't show in the record that you're signed up for that particular quest.";
		mes "Sorry, I can't do anything about it!";
		close;
	}

	function F_erase_quest_range {
		set .@lower_range, getarg(0);
		set .@upper_range, getarg(1);
		//mes ".@lower_range: "+ .@lower_range;
		//mes ".@upper_range: "+ .@upper_range;
		//next;
		for(set .@i,.@lower_range; .@i <= .@upper_range; set .@i,.@i+1) {
			//mes "checking quest" + .@i;
			//next;
			if(isbegin_quest(.@i)) {
				clear;
				//mes "erasing quest #" + .@i;
				erasequest .@i;
				//next;
				continue;
			}
		}
		return;
	}
	
	function F_erase_quest_array { // pass in an array of quest IDs to have them all removed from ongoing quests
		for(set .@i,0; .@i < getarraysize(getarg(0)); set .@i,.@i+1) {
			if(isbegin_quest(getelementofarray(getarg(0),.@i))) {
				erasequest getelementofarray(getarg(0),.@i);
				continue;
			}
		}
		return;
	}
	
	function F_compile_board_quests { // Input the str name of the board to add its quests to .every_board_quest_id array
		for(set .@i,0; .@i < getarraysize(getvariableofnpc(.all_quest_ids,getarg(0))); set .@i,.@i+1) {
			set .every_board_quest_id[.every_board_quest_id_index],getvariableofnpc(.all_quest_ids[.@i],getarg(0));
			set .every_board_quest_id_index,.every_board_quest_id_index+1;
		}
	}
