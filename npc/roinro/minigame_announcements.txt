/*
Every 2 hours, a minigame NPC will appear in Prontera. This script sends announcements 30/15/5 minutes before and enables/disables said NPCs
*/

-	script	Announce Minigames	-1,{
OnInit:
	setarray .minigame_order$[0],
		"Whack-a-Poring",
		"Poring Sequence",
		"The Enigmatic Slot Machine",
		"Poring Rancher",
		"Roin Coin Black Jack";
	if($next_minigame > 4) {
		$next_minigame -= 5;
	}
	setarray .npc_name$[0],
		"Whack-a-Poring#Prontera",
		"Poring Sequence#Prontera",
		"Enigmatic Slot Machine",
		"Poring Rancher#Prontera",
		"Black Jack Dealer#Prontera";
		
	function looparound_value { // Args: [0] input, [1] increase input by this, [2] don't loop above this value, [3] don't loop below this value
		if(getarg(0) + getarg(1) > getarg(2)) {
			return getarg(3);
		}
		if(getarg(0) + getarg(1) < getarg(3)) {
			return getarg(2);
		}
		return getarg(0) + getarg(1);
	}
	end;

// Every 2 hours, a new event happens. First, announce 30 minutes before
OnClock0130:
OnClock0330:
OnClock0530:
OnClock0730:
OnClock0930:
OnClock1130:
OnClock1330:
OnClock1530:
OnClock1730:
OnClock1930:
OnClock2130:
OnClock2330:
	announce .minigame_order$[$next_minigame] + " will take place 30 minutes from now in Prontera. Don't miss out!", bc_all;
	end;
// Announce 15 minutes before
OnClock0145:
OnClock0345:
OnClock0545:
OnClock0745:
OnClock0945:
OnClock1145:
OnClock1345:
OnClock1545:
OnClock1745:
OnClock1945:
OnClock2145:
OnClock2345:
	announce .minigame_order$[$next_minigame] + " will start in 15 minutes in Prontera!", bc_all;
	end;
// Announce 5 minutes before
OnClock0155:
OnClock0355:
OnClock0555:
OnClock0755:
OnClock0955:
OnClock1155:
OnClock1355:
OnClock1555:
OnClock1755:
OnClock1955:
OnClock2155:
OnClock2355:
	announce .minigame_order$[$next_minigame] + " will be here in just 5 minutes! Come to Prontera to play!", bc_all;
	end;
// Annouce event + unhide NPCs
OnClock0200:
OnClock0400:
OnClock0600:
OnClock0800:
OnClock1000:
OnClock1200:
OnClock1400:
OnClock1600:
OnClock1800:
OnClock2000:
OnClock2200:
OnClock0000:
	// Slot machine only lasts 5 minutes
	if (.minigame_order$[$next_minigame] == "The Enigmatic Slot Machine") {
		announce .minigame_order$[$next_minigame] + " is now happening in Prontera! It will be here for 5 minutes.", bc_all;
		hideoffnpc .npc_name$[$next_minigame];
		set getvariableofnpc(.isclosed,"Enigmatic Slot Machine"), 0;
		sleep 300000;
	}
	//Everything else lasts 15 minutes
	else {
		announce .minigame_order$[$next_minigame] + " is now happening in Prontera! It will be here for 15 minutes.", bc_all;
		hideoffnpc .npc_name$[$next_minigame];
		sleep 600000; // Sleep for 10 minutes
		announce .minigame_order$[$next_minigame] + " will be ending in 5 minutes.", bc_all;
		sleep 300000; // 5 minute sleep
	}
	announce .minigame_order$[$next_minigame] + " has left Prontera. The next event will be " +
	.minigame_order$[looparound_value($next_minigame, 1, getarraysize(.minigame_order$)-1, 0)] + 
	", which will take place in just under 2 hours!", bc_all;
	hideonnpc .npc_name$[$next_minigame];
	set getvariableofnpc(.isclosed,"Enigmatic Slot Machine"), 1;
	$next_minigame++;
	if($next_minigame > 4) {
		$next_minigame -= 5;
	}
	end;
}
