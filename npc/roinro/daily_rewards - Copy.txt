-	script	DailyLogin	-1,{
	
	OnPCLoginEvent:
	if(#DailyRewardTime > gettimetick(2)) {
		dispbottom .n$+": You can't currently receive daily rewards; you still have to wait "+callfunc("Time2Str",#DailyRewardTime);
		end;
	} else {
		set #DailyRewardTime,0;
		set #already_received_dailies,0;
		set .@d_login,getvariableofnpc(.rew_login,"DailyLogin")-#daily;
		if(.@d_login == 0 || .times_up == 1){
			.times_up = 1;
			announce .n$+": You can claim your daily rewards! Please stop by the Main Office to receive them.",bc_self;
		} else {
		//.times_up = 0;
		announce .n$+": Stay active for "+((.@d_login%60 == 0)?.@d_login/60+" hour(s)":.@d_login+" minute")+((.@d_login == 1)?"":"s")+" to receive daily rewards!",bc_self;
		}
		addtimer .rew_timer,"DailyLogin::OnTimeCheck";
		end;

		OnTimeCheck:
		deltimer "DailyLogin::OnTimeCheck";
		if(checkidle() <= .idle) {
			if(@CheckIdle) 
				set @CheckIdle,0;
			set #daily,#daily + .rew_count;
			if(#daily == .rew_login) {
				set #daily,0;
				if(.rew_type == 2)
					set #DailyReward,DT_DAYOFMONTH;
				else 
					set #DailyReward,#Daily2 + 1;
				.times_up = 1;
				announce .n$+": You can now claim daily rewards! Please visit the Main Office in Prontera to receive it.",bc_self;
				end;
			}
			
		} else {
			if(!@CheckIdle) {
				announce .n$+": You need to stay active for your Daily Reward. I will check again in "+.rew_count+" minute(s).",bc_self;
				set @CheckIdle,1;
			}
		}
		addtimer .rew_timer,"DailyLogin::OnTimeCheck";
		end;
	}
	OnInit:
	set .n$,"[Daily Rewards]";
	set .rew_time,81000; // Timer until next Reward in seconds (22.5 hours)
	// Check in seconds how long the player can be "idle"
	set .idle,1680; // (28 minutes)
	// How long does someone have to stay online to recieve the reward in minutes
	// Default: 60 = 1 Hour
	set .rew_login,30;
	// How many minutes should pass for the counter to increase > OnTimeCheck
	set .rew_count,1;
	// Dynamic Timer - Don't touch this please
	set .rew_timer,.rew_count*60*1000;
	// Reward Type:
	// 1 = Continues Reward
	// 2 = Depending on day
	set .rew_type,1;
	end;
}
