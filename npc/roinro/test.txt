// Save the player's current stat build
// Command: /savebuild
prontera,150,150,4	script	StatUpDown	123,{
	mes "STR: 20","VIT: 25","AGI: 30","INT: 35","DEX: 45","LUK: 10","POW: 0","STA: 1";
	switch(select("YES:NO")) {
		case 1:
			resetstatus;
			// STR
			for(i = 1; i < 20; i++){
				statusup bStr;
			}
			// VIT
			for(i = 1; i < 25; i++){
				statusup bVit;
			}
			// AGI
			for(i = 1; i < 30; i++){
				statusup bAgi;
			}
			// INT
			for(i = 1; i < 35; i++){
				statusup bInt;
			}
			// DEX
			for(i = 1; i < 45; i++){
				statusup bDex;
			}
			// LUK
			for(i = 1; i < 10; i++){
				statusup bLuk;
			}
			// POW
			for(i = 0; i < 0; i++){
				traitstatusup bPow;
			}
			// STA
			for(i = 0; i < 1; i++){
				traitstatusup bSta;
			}
			break;
		case 2:
			break;
	}
	end;
}

prontera,157,150,3,	script	Effect Testing NPC	982,{
	if (getgmlevel() < 99) end;
	mes "Enter an effect number from 0-967", "Type DONE or X when finished";
EffectInputLoop:
	input .@userinput$;
	.@userinput = .@userinput$;
	if(strtolower(.@userinput$) == "done" || strtolower(.@userinput$) == "x") {
		clear;
		mes "Closing NPC script...";
		close;
	}
	if(.@userinput < 0 || .@userinput > 967) {
		clear;
		mes "Not a valid input.", "Enter an effect number from 0-967", "Type DONE or X when finished";
		goto EffectInputLoop;
	}
	specialeffect .@userinput;
	specialeffect2 .@userinput;
	//misceffect .@userinput;
	goto EffectInputLoop;
}
