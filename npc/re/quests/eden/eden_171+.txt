			mes "[Mission Board]";
			mes "You have completed the ^4d4dff"+.@quest_names$[.@i]+" ^000000mission.";
			mes "Would you like to turn it in now?";
				if (select("Heck yeah!:Not yet") == 2) {close;}
			erasequest .all_quest_ids[.@i];
			specialeffect2 EF_STEAL;
			getexp .@Exp[.@i],.@Exp[.@i];
			next;
		}
		
	if (BaseLevel < 171) {
		mes "This mission bulletin board is for Lv171+ adventurers.";
		close;
	}
	mes "[Mission Board]";
	mes "A bunch of missions are listed on the board.";
	mes "Choose what location you'd like to hunt at:";
	next;
	switch(select("Tomb of the Fallen:Cancel")) {
	/*case :
		switch(select("Chaotic Poporing Hunting:Chaotic Mantis Hunting:Chaotic Hunter Fly Hunting:Chaotic Side Winder Hunting:Chaotic Killer Mantis Hunting")) {
			case 1: callsub Accept_Quest,100016,"Hunt 30 ^4d4dffChaotic Poporing^000000:";

	case 1:

			case 4: callsub Accept_Quest,100024,"Hunt 30 ^4d4dffRanger Cecil Damon^000000:";

						mes "Yeah, maybe it'd be too much.";

	case 2:

Accept_Quest:
	clear;
	mes "[Mission Board]";
	for ( .@i = 1; .@i < getargcount(); .@i++ )
		mes getarg(.@i);
	next;
	if(select("Accept the mission:Cancel") == 1) {
		if (checkquest(getarg(0)) > -1) {mes "You are already signed up for this mission!"; close;};
		mes "[Mission Board]";
		mes "You have accepted the mission. For more details, please press 'alt + u' keys, you can get information from the Quest Window.";
		setquest getarg(0);
	}
	close;
	OnInit: {
		setarray .all_quest_ids[0],
			// Illusion of Labyrinth
			//100016,100017,100018,100019,100020, these monsters don't exist yet
