moc_para01,36,38,3	script	Mission [11 - 25]	857,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "You are not an Eden group member. You are not qualified to access the bulletin board.";
		close;
	}
	
	setarray .@quest_name$[0],
		"Hornet Hunting","Condor Hunting","Rocker Hunting","Worm Tail Hunting",
		"Spore Hunting","Thief Bug Egg Hunting","Muka Hunting",
		"Familiar Hunting","Peco Peco Hunting","Poison Spore Hunting";
	setarray .@item_reqs[0],
		0,0,940,0,0,915,0,0,0,7033;

	setarray .@item_amounts[0],
		0,0,10,0,0,10,0,0,0,5;

	setarray .@Exp[0],
		900,1800,1400,2600,
		3900,2900,3200,
		3200,5400,6000;
	for(set .@i,0; .@i<getarraysize(.@Exp); set .@i,.@i+1)
		if (checkquest(.all_quest_ids[.@i],HUNTING) == 2) { // Check if the hunting part of the quest is completed
			// Now check if item gathering is complete
			.@item_req = .@item_reqs[.@i];
			.@req_amount = .@item_amounts[.@i];
			if (.@item_req) { // Check if the gathering part of the quest is completed
				if (countitem(.@item_req) < .@req_amount) {
					mes "You don't have enough items to turn in a quest.";
					mes "You need to bring "+ .@req_amount +" "+ getitemname(.@item_req) +".";
					next;
					continue;
				}
			}
			mes "[Mission Board]";
			mes "You have completed the ^4d4dff"+.@quest_name$[.@i]+" ^000000mission.";
			//mes "Here is your reward.";
			mes "Would you like to turn it in now?";
				if (select("Heck yeah!:Not yet") == 2) {close;}
			//erasequest 12088;
			erasequest .all_quest_ids[.@i];
			specialeffect2 EF_STEAL;
			getexp .@Exp[.@i],.@Exp[.@i];
			if (.@item_req)
				delitem .@item_req,.@req_amount;
			next;
		}
		
	if (checkquest(12088) == -1) {
		if (BaseLevel < 11 || BaseLevel > 25) {
			mes "Mission bulletin board for Lv11~25 adventurers.";
			close;
		}
		mes "[Mission Board]";
		mes "A bunch of missions are listed on the board.";
		next;
		if(select("Check the Missions:Cancel") == 2) {
			mes "^000077Ok. Let's check the missions next time.^000000.";
			close;
		}
		switch(select("Hornet Hunting:Condor Hunting:Rocker Hunting:Worm Tail Hunting:Spore Hunting:Thief Bug Egg Hunting:Muka Hunting:Familiar Hunting:Peco Peco Hunting:Poison Spore Hunting:Cancel")) {
			case 1: callsub L_Quest,11114,"Hunt 10 ^4d4dffHornets^000000:";
			case 2: callsub L_Quest,11115,"Hunt 20 ^4d4dffCondors^000000:";
			case 3: callsub L_Quest,11116,"Hunt 10 ^4d4dffRockers^000000","Bring back 10 ^4d4dffGrasshopper's Legs^000000:";
			case 4: callsub L_Quest,11117,"Hunt 20 ^4d4dffWorm Tails^000000:";
			case 5: callsub L_Quest,11118,"Hunt 30 ^4d4dffSpores^000000:";
			case 6: callsub L_Quest,11119,"Hunt 20 ^4d4dffThief Bug Eggs^000000","Bring back 10 ^4d4dffChrysalis^000000:";
			case 7: callsub L_Quest,11120,"Hunt 20 ^4d4dffMuka^000000:";
			case 8: callsub L_Quest,11121,"Hunt 20 ^4d4dffFamiliar^000000:";
			case 9: callsub L_Quest,11122,"Hunt 30 ^4d4dffPeco Pecos^000000:";
			case 10: callsub L_Quest,11123,"Hunt 30 ^4d4dffPoison Spores^000000","Bring back 10 ^4d4dffPoison Spores^000000 (item):";
		}
		end;
	}
	
	if ((checkquest(11114,HUNTING) == 0) && (checkquest(11115,HUNTING) == 0) && (checkquest(11116,HUNTING) == 0) && (checkquest(11117,HUNTING) == 0) && (checkquest(11118,HUNTING) == 0) && (checkquest(11119,HUNTING) == 0) && (checkquest(11120,HUNTING) == 0) && (checkquest(11121,HUNTING) == 0) && (checkquest(11122,HUNTING) == 0) && (checkquest(11123,HUNTING) == 0)) {
		mes "You may now view the bulletin board.";
		erasequest 12088;
		close;
	}
	mes "There is nothing special to do.";
	close;
L_Quest:
	clear;
	mes "[Mission Board]";
	for ( .@i = 1; .@i < getargcount(); .@i++ )
		mes getarg(.@i);
	next;
	if(select("Accept the mission:Cancel") == 1) {
		if (checkquest(getarg(0)) > -1) {mes "You are already signed up for this mission!"; close;};
		mes "[Mission Board]";
		mes "You have accepted the mission. For more details, please press 'alt + u' keys, you can get information from the Quest Window.";
		//setquest 12088; // Commenting this out to allow taking multiple quests
		setquest getarg(0);
	}
	close;
	OnInit: {
		setarray .all_quest_ids[0], //used for quest deletion
			11114,11115,11116,11117,
			11118,11119,11120,
			11121,11122,11123;
	}
}
