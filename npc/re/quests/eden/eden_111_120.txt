// Roin's 111-120 bounty board
// Last edited 4/21/2025

moc_para01,48,181,3	script	111-120 Mission Board	857,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "You are not an Eden group member. You are not qualified to access the bulletin board.";
		close;
	}
	
	//Hunting Quest Turn-ins first
	setarray .@hunting_names$[0],
		"Raydric Hunting","Khalitzburg Hunting","Wanderer Hunting",
		"Ancient Mimic Hunting","Deathword Hunting","Owl Baron Hunting",
		"Dark Pinguicula Hunting","Nepenthes Hunting","Naga Hunting", "Centipede Larva Hunting", "Cornus Hunting",
		"Ancient Mummy Hunting","Geffenia Hunting","Venatu Hunting";
	setarray .@hunting_ids[0],
		13067,13069,13071,
		13073,13075,13077,
		13081,13083,13085,13087,13089,
		13093,13095,13097;
	setarray .@hunting_exp[0], // https://irowiki.org/wiki/Base_EXP_Chart (amount of exp it takes to level up from 150) / 25 * 1.85
		350000,350000,350000,
		350000,350000,350000,
		350000,350000,350000,300000,350000,
		350000,450000,1050000;
	for(set .@i,0; .@i<getarraysize(.@hunting_ids); set .@i,.@i+1)
		if (checkquest(.@hunting_ids[.@i],HUNTING) == 2) { // Check if the hunting part of the quest is completed
			mes "[Mission Board]";
			mes "You have completed the ^4d4dff"+.@hunting_names$[.@i]+" ^000000mission.";
			mes "Would you like to turn it in now?";
				if (select("Heck yeah!:Not yet") == 2) {close;}
			erasequest .@hunting_ids[.@i];
			specialeffect2 EF_STEAL;
			getexp .@hunting_exp[.@i],.@hunting_exp[.@i];
			next;
		}
	//Gathering Quest Turn-ins second
	setarray .@gathering_names$[0],
		"Gather Bloody Pages","Gather Mystic Horns","Gather Fragments & Screws";
	setarray .@gathering_ids[0],
		13079,13091,13099;
	setarray .@gathering_exp[0],
		175000,175000,175000;
	setarray .@gathering_reqs[0],
		7449,6023,7094;
	setarray .@gathering_amounts[0],
		20,20,10;
	setarray .@gathering_reqs2[0],
		0,0,7317;
	setarray .@gathering_amounts2[0],
		0,0,20;
	for(.@i = 0; .@i<3; .@i++)
		if(checkquest(.@gathering_ids[.@i]) > -1) {
			if(.@i == 2) { //if the index is 2, which is the only quest requiring multiple item turn-ins
				.@number_of_items = 2;
			} else {
				.@number_of_items = 1;
			}
			if(.@number_of_items == 1) {
				debugmes "Only counting 1 item, sir.";
				.@item_req = .@gathering_reqs[.@i];
				.@req_amount = .@gathering_amounts[.@i];
				if (countitem(.@item_req) < .@req_amount) {
					mes "You don't have enough items to turn in a quest.";
					mes "You need to bring "+ .@req_amount +" ^4d4dff"+ getitemname(.@item_req) +"^000000.";
					next;
					continue;
				}
				mes "[Mission Board]";
				mes "You have completed the ^4d4dff"+.@gathering_names$[.@i]+" ^000000mission.";
				mes "Would you like to turn it in now?";
					if (select("Heck yeah!:Not yet") == 2) {close;}
				erasequest .@gathering_quest_ids[.@i];
				specialeffect2 EF_STEAL;
				getexp .@gathering_exp[.@i],(.@gathering_exp[.@i]/2);
				if (.@item_req)
					delitem .@item_req,.@req_amount;
				next;
			} else if(.@number_of_items == 2) {
				debugmes "Counting 2 items, sir.";
				.@item_req1 = .@gathering_reqs[.@i];
				.@req_amount1 = .@gathering_amounts[.@i];
				.@item_req2 = .@gathering_reqs2[.@i];
				.@req_amount2 = .@gathering_amounts2[.@i];
				if (countitem(.@item_req1) < .@req_amount1 || countitem(.@item_req2) < .@req_amount2) {
					mes "You don't have enough items to turn in a quest.";
					mes "You need to bring "+ .@req_amount1 +" ^4d4dff"+ getitemname(.@item_req1) +"^000000 and " + .@req_amount2 + " ^4d4dff" + getitemname(.@item_req2) + "^000000.";
					next;
					continue;
				}
				mes "[Mission Board]";
				mes "You have completed the ^4d4dff"+.@gathering_names$[.@i]+" ^000000mission.";
				mes "Would you like to turn it in now?";
					if (select("Heck yeah!:Not yet") == 2) {close;}
				erasequest .@gathering_ids[.@i];
				specialeffect2 EF_STEAL;
				getexp .@gathering_exp[.@i],(.@gathering_exp[.@i]/2);
				delitem .@item_req1,.@req_amount1;
				delitem .@item_req2,.@req_amount2;
				next;
			}
		}
	
	//Open mission board lastly
	if (BaseLevel < 111 || BaseLevel > 120) {
		mes "Mission bulletin board for Lv111~120 adventurers.";
		close;
	}
	mes "[Mission Board]";
	mes "A bunch of missions are listed on the board.";
	switch(select("Hunting Missions:Gathering Missions:Cancel")) {
	case 1:
		clear;
		mes "[Mission Board]";
		mes "Choose what location you'd like to hunt at:";
		switch(select("Glast Heim:Thanatos Tower:Ash Vacuum:Other Areas")) {
		case 1:
			setarray .@all_quests_in_area[0],
				13067, 13069, 13071;
			switch(select("Raydric Hunting:Khalitzburg Hunting:Wanderer Hunting:Hunt ALL")) {
				case 1: callsub Accept_Quest,13067,"Hunt 30 ^4d4dffRaydrics^000000:";
				case 2: callsub Accept_Quest,13069,"Hunt 30 ^4d4dffKhalitzburgs^000000:";
				case 3: callsub Accept_Quest,13071,"Hunt 30 ^4d4dffWanderers^000000:";
				case 4:
					mes "You would like to hunt ALL of these monsters?";
					switch(select("Yes:Never mind")) {
						case 1:
							for(set .@i,0 ; .@i < getarraysize(.@all_quests_in_area) ; set .@i,.@i+1) {
								if (checkquest(.@all_quests_in_area[.@i]) == -1) {
									setquest .@all_quests_in_area[.@i];
								}
								continue;
							}
							mes "You have signed up for all of those quests.";
							close;
						case 2:
							mes "Yeah, maybe it'd be too much.";
							close;
					}
					break;
			}
		case 2:
			setarray .@all_quests_in_area[0],
				13073, 13075, 13077;
			switch(select("Ancient Mimic Hunting:Deathword Hunting:Owl Baron Hunting:Hunt ALL")) {
				case 1: callsub Accept_Quest,13073,"Hunt 30 ^4d4dffAncient Mimics^000000:";
				case 2: callsub Accept_Quest,13075,"Hunt 30 ^4d4dffDeathwords^000000:";
				case 3: callsub Accept_Quest,13077,"Hunt 20 ^4d4dffOwl Barons^000000:";
				case 4:
					mes "You would like to hunt ALL of these monsters?";
					switch(select("Yes:Never mind")) {
						case 1:
							for(set .@i,0 ; .@i < getarraysize(.@all_quests_in_area) ; set .@i,.@i+1) {
								if (checkquest(.@all_quests_in_area[.@i]) == -1) {
									setquest .@all_quests_in_area[.@i];
								}
								continue;
							}
							mes "You have signed up for all of those quests.";
							close;
						case 2:
							mes "Yeah, maybe it'd be too much.";
							close;
					}
					break;
			}
		case 3:
			setarray .@all_quests_in_area[0],
				13081, 13083, 13085, 13087, 13089;
			switch(select("Dark Pinguicula Hunting:Nepenthes Hunting:Naga Hunting:Centipede Larva Hunting:Cornus Hunting:Hunt ALL")) {
				case 1: callsub Accept_Quest,13081,"Hunt 30 ^4d4dffDark Pinguiculas^000000:";
				case 2: callsub Accept_Quest,13083,"Hunt 30 ^4d4dffNepenthes^000000:";
				case 3: callsub Accept_Quest,13085,"Hunt 30 ^4d4dffNaga^000000:";
				case 4: callsub Accept_Quest,13087,"Hunt 20 ^4d4dffCentipede Larvae^000000:";
				case 5: callsub Accept_Quest,13089,"Hunt 30 ^4d4dffCornus^000000:";
				case 6:
					mes "You would like to hunt ALL of these monsters?";
					switch(select("Yes:Never mind")) {
						case 1:
							for(set .@i,0 ; .@i < getarraysize(.@all_quests_in_area) ; set .@i,.@i+1) {
								if (checkquest(.@all_quests_in_area[.@i]) == -1) {
									setquest .@all_quests_in_area[.@i];
								}
								continue;
							}
							mes "You have signed up for all of those quests.";
							close;
						case 2:
							mes "Yeah, maybe it'd be too much.";
							close;
					}
					break;
			}
		case 4:
			setarray .@all_quests_in_area[0],
				13093, 13095, 13097;
			switch(select("Ancient Mummy Hunting:Geffenia Hunting:Venatu Hunting:Hunt ALL")) {
				case 1: callsub Accept_Quest,13093,"Hunt 30 ^4d4dffAncient Mummies^000000:";
				case 2: callsub Accept_Quest,13095,"Hunt 10 ^4d4dffIncubus^000000","Hunt 10 ^4d4dffSucubus^000000","Hunt 10 ^4d4dffVioly^000000:";
				case 3: callsub Accept_Quest,13097,"Hunt 30 ^2D8E2DGreen Venatu^000000","Hunt 30 ^ff6a00Orange Venatu^000000","Hunt 30 ^4d4dffBlue Venatu^000000:";
			}
		case 5:
			
		}
	case 2:
		clear;
		mes "[Mission Board]";
		mes "Choose which item(s) you'd like to gather:";
		switch(select("Bloody Pages:Mystic Horns:Fragments & Screws")) {
			case 1: callsub Accept_Quest,13079,"Gather 20 ^4d4dffBloody Pages^000000:";
			case 2: callsub Accept_Quest,13091,"Gather 20 ^4d4dffMystic Horns^000000:";
			case 3: callsub Accept_Quest,13099,"Gather 10 ^4d4dffFragments^000000","Gather 20 ^4d4dffRusty Screws^000000:";
		}
	case 3:
		end;
	}
		
	mes "There is nothing special to do.";
	close;

	
Accept_Quest:
	clear;
	mes "[Mission Board]";
	for ( .@i = 1; .@i < getargcount(); .@i++ )
		mes getarg(.@i);
	next;
	if(select("Accept the mission:Cancel") == 1) {
		if (checkquest(getarg(0)) > -1) {mes "You are already signed up for this mission!"; close;};
		mes "[Mission Board]";
		mes "You have accepted the mission. For more details, please press 'alt + u' keys, you can get information from the Quest Window.";
		setquest getarg(0);
	}
	close;
	OnInit: {
		setarray .all_quest_ids[0],
			13067,13069,13071,
			13073,13075,13077,
			13081,13083,13085,13087,13089,
			13093,13095,13097,
			13079,13091,13099;
	}
}
